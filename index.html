function parseAndLoadStudy() { let rawText = aiInput.value.trim(); if (!rawText) { alert("Please paste the AI word list data first!"); return; } displayBox.innerHTML = ""; activeWordToken = null; const fullRefText = sentenceInput.value.trim(); if(fullRefText) { sentenceBox.textContent = fullRefText; toggleBtn.style.display = "block"; } else { toggleBtn.style.display = "none"; } // --- FIXED ROBUST SPLITTER --- // Finds the note's punctuation (like "match.") followed by space, right before the next item begins. rawText = rawText.replace(/([\.）\?\.!])\s+(?=[^|\n]+?\|)/g, "$1\n"); // Split into clean lines const lines = rawText.split('\n').map(l => l.trim()).filter(l => l !== ""); const dataSegments = []; lines.forEach(line => { const parts = line.split('|').map(p => p.trim()); // Read up to 4 potential items per split block safely if (parts.length >= 3) { const chinese = parts[0]; const pinyin = parts[1]; const english = parts[2]; const note = parts[3] ? parts[3] : ""; if (chinese) { dataSegments.push({ chinese, pinyin, english, note }); } } }); if (dataSegments.length === 0) { alert("Error: The text format couldn't be automatically extracted. Please check your data contains '|' dividers."); return; } dataSegments.forEach(data => { let displayPinyin = data.pinyin === "—" ? "" : data.pinyin; let displayEnglish = data.english === "—" ? "Not directly translated" : data.english; let displayNote = data.note === "—" ? "" : data.note; // Strip any accidental curly quotes provided around translation blocks displayEnglish = displayEnglish.replace(/^["“'‘]|["”'’]$/g, ''); const token = document.createElement('span'); token.className = 'word-token state-chinese'; token.textContent = data.chinese; // CHANGED: If a note exists (even if it says "Direct match"), allow the 4th state! const hasValidNote = (displayNote !== ""); const maxStates = hasValidNote ? 4 : 3; let currentState = 0; token.triggerCycle = () => { currentState = (currentState + 1) % maxStates; token.className = 'word-token' + (token === activeWordToken ? ' active-word' : ''); if (currentState === 0) { token.textContent = data.chinese; token.classList.add('state-chinese'); } else if (currentState === 1) { token.textContent = displayPinyin || "—"; token.classList.add('state-pinyin'); } else if (currentState === 2) { token.textContent = displayEnglish; token.classList.add('state-english'); } else if (currentState === 3) { token.textContent = displayNote; token.classList.add('state-note'); } }; token.onclick = (e) => { e.stopPropagation(); e.preventDefault(); if (activeWordToken === token) { token.triggerCycle(); } else { if (activeWordToken) { activeWordToken.classList.remove('active-word'); } activeWordToken = token; token.classList.add('active-word'); } }; displayBox.appendChild(token); displayBox.appendChild(document.createTextNode(" ")); }); document.getElementById('input-page').classList.remove('active'); document.getElementById('study-page').classList.add('active'); document.getElementById('nav-bar').style.display = 'flex'; sentenceBox.style.display = "none"; displayBox.style.display = "block"; }
